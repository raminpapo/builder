Object.keys(exposePackages).forEach(packageName => {
  system.register(packageName, [], _export => {
    return {
      execute: () => {
        const content = (exposePackages as any)[packageName];
        _export({
          ...content,
          default: content,
        });
      },
    };
  });
});
